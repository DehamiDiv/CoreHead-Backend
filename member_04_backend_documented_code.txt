// 2. DEVELOPED /ai/generate-layout ENDPOINT & 5. SECURITY VALIDATION
// Location: src/routes/aiRoutes.js
const express = require('express');
const router = express.Router();
const rateLimit = require('express-rate-limit');

const aiLimiter = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 mins
  max: 10, // Limit each IP to 10 requests per window
  message: { error: 'Too many AI requests, please try again later.' }
});

router.post('/generate-layout', aiLimiter, async (req, res) => {
  const { prompt } = req.body;

  // Security: Prompt sanitization & Length validation
  if (!prompt || prompt.trim().length < 5) {
    return res.status(400).json({ error: 'Invalid prompt length' });
  }
  if (prompt.toLowerCase().includes('<script>')) {
    return res.status(400).json({ error: 'Script injection detected' });
  }

  const result = await aiService.generateLayout(prompt.trim());
  res.json({ success: true, ...result });
});
